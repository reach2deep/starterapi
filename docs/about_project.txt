rules:
  # Naming Conventions
  - name: "Class Naming"
    pattern: "*.cs"
    rules:
      - "Interfaces should start with 'I'"
      - "DTOs should end with 'Request' or 'Response'"
      - "Controllers should end with 'Controller'"
      - "Services should end with 'Service'"
      - "Repositories should end with 'Repository'"
      - "Database contexts should end with 'DbContext'"
      - "Mapping profiles should end with 'MappingProfile'"

  # Module Organization
  - name: "Module Structure"
    description: "Rules for module-based organization"
    rules:
      - "Each module should be under Modules/[Global|Tenant]/[ModuleName]"
      - "Each module should contain its own DTOs, Services, Interfaces, and Mappings"
      - "Module-specific entities should be in Core/Modules/[Global|Tenant]/[ModuleName]/Entities"
      - "Cross-module dependencies should be minimized"
      - "Common code should be in the Common module"

  # Architecture Rules
  - name: "Layer Dependencies"
    description: "Enforce clean architecture dependencies"
    rules:
      - "Core layer should not depend on other layers"
      - "Application layer can only depend on Core layer"
      - "Infrastructure layer can depend on Core and Application layers"
      - "API layer can depend on Application layer only"
      - "Shared layer can be used by all layers"

  # Multi-tenancy Rules
  - name: "Multi-tenancy Implementation"
    pattern: "*.cs"
    rules:
      - "Global modules must be in Modules/Global namespace"
      - "Tenant modules must be in Modules/Tenant namespace"
      - "TenantId must be included in all tenant-specific entities"
      - "Use TenantMiddleware for tenant resolution"
      - "Tenant context must be validated in all tenant-specific operations"

  # Code Organization
  - name: "Project Structure"
    folders:
      - src/starterkit.API:
          - "Controllers/Modules/V1/Global for global endpoints"
          - "Controllers/Modules/V1/Tenant for tenant-specific endpoints"
          - "DependencyInjection for service registration"
          - "Middlewares must include TenantMiddleware"
      
      - src/starterkit.Core:
          - "Modules/[Global|Tenant] for domain entities and logic"
          - "Modules/Common for shared entities"
          - "Interfaces/Repositories for repository contracts"
      
      - src/starterkit.Application:
          - "Modules/[Global|Tenant]/[ModuleName] for module implementation"
          - "Each module must have DTOs, Interfaces, Services, Validators, and Mappings"
          - "DependencyInjection for service registration"
      
      - src/starterkit.Infrastructure:
          - "Persistence/RootDb and Persistence/TenantDb for database contexts"
          - "Services for infrastructure services"
          - "Repositories/[Global|Tenant] for repository implementations"

  # Module Implementation Rules
  - name: "Module Implementation"
    pattern: "*/Modules/*/*.cs"
    rules:
      - "Each module should be self-contained"
      - "Use interfaces for cross-module communication"
      - "Keep module-specific logic within the module"
      - "Follow consistent folder structure within modules"
      - "Document module dependencies and requirements"

  # Database Rules
  - name: "Database Contexts"
    pattern: "*/Persistence/*DbContext.cs"
    rules:
      - "RootDbContext for global operations"
      - "TenantDbContext for tenant-specific operations"
      - "Use separate connection strings per tenant"
      - "Implement proper tenant data isolation"
      - "Use DbContextFactory for context creation"

  # Authentication Rules
  - name: "Authentication"
    pattern: "*/Auth/*.cs"
    rules:
      - "Implement two-step authentication (global then tenant)"
      - "Include tenant information in JWT tokens"
      - "Validate tenant access in authorization"
      - "Handle tenant-specific refresh tokens"
      - "Keep authentication logic in Global/Auth module"

  # Service Implementation
  - name: "Services"
    pattern: "*/Services/*.cs"
    rules:
      - "Services must be in module's Services folder"
      - "Implement corresponding interface from module's Interfaces folder"
      - "Use dependency injection for dependencies"
      - "Handle module-specific validation and business logic"
      - "Use appropriate error handling and logging"

  # Repository Rules
  - name: "Repositories"
    pattern: "*/Repositories/*.cs"
    rules:
      - "Implement repository interfaces from Core layer"
      - "Use appropriate DbContext (Root or Tenant)"
      - "Ensure proper data isolation"
      - "Handle tenant context in tenant repositories"
      - "Implement proper error handling"

  # API Endpoints
  - name: "API Controllers"
    pattern: "*/Controllers/*.cs"
    rules:
      - "Place in appropriate module folder under Controllers/Modules/V1/[Global|Tenant]/[ModuleName]/[ControllerName]"
      - "Inherit from BaseApiController"
      - "Use module-specific DTOs"
      - "Proper error handling and response codes"
      - "Document API endpoints"

  # Error Handling
  - name: "Error Handling"
    pattern: "*.cs"
    rules:
      - "Use consistent error handling across modules"
      - "Proper exception handling in services"
      - "Clear error messages and logging"
      - "Include relevant context in errors"
      - "Use appropriate HTTP status codes"

  # Testing
  - name: "Testing"
    pattern: "*/Tests/*.cs"
    rules:
      - "Organize tests by module structure"
      - "Test module-specific functionality"
      - "Mock dependencies appropriately"
      - "Test both success and failure scenarios"
      - "Follow consistent testing patterns"

  # Security
  - name: "Security"
    pattern: "*.cs"
    rules:
      - "Proper authentication and authorization"
      - "Secure handling of sensitive data"
      - "Input validation in DTOs and validators"
      - "Proper tenant isolation"
      - "Secure configuration management"

- Use consistent naming conventions for all services
